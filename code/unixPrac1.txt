#1 count number of lines
## --total=never stops sum from being calculated
wc -l ../data/fasta/* --total=never

#2 print everything starting from the second line
tail -n +2 ../data/fasta/e_coli.fasta
# header is excluded as it is not part of the sequence
# -n specifies the number of lines to print. +2 tells it to omit the first 2 lines

#3 count the sequence length
tail -n +2 ../data/fasta/e_coli.fasta |  wc -c 

#4 count the matches of ATGC
tail -n +2 ../data/fasta/e_coli.fasta | grep -o ATGC | wc -l
# grep -o finds each match, the pipe passes this to a line count

#5 calculate AT/GC ratio
echo "scale=4; ($(tail -n +2 ../data/fasta/e_coli.fasta | grep -o A | wc -l) + $(tail -n +2 ../data/fasta/e_coli.fasta | grep -o T | wc -l)) / ($(tail -n +2 ../data/fasta/e_coli.fasta | grep -o C | wc -l) + $(tail -n +2 ../data/fasta/e_coli.fasta | grep -o G | wc -l))" | bc
##scale sets the number of decimal points
