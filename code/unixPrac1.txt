#1 counts number of lines in each .fasta file in the data folder
## --total=never stops sum from being calculated
wc -l ../data/fasta/* --total=never

#2 print everything starting from the second line in the e_coli.fasta file
tail -n +2 ../data/fasta/e_coli.fasta
# header is excluded as it is not part of the sequence
# -n specifies the number of lines to print. +2 tells it to omit the first 2 lines

#3 count the number of characters in the e_coli sequence. Pipe previous command into a tr which deletes line breaks, then pipe into a word count command
tail -n +2 ../data/fasta/e_coli.fasta |tr -d '\n'  | wc -m

#4 count the matches of ATGC. Pipe as in #3 but grep term searches for ATGC in the file and wc -l counts the number of grep results
tail -n +2 ../data/fasta/e_coli.fasta |tr -d '\n'  | grep -o ATGC | wc -l

#5 calculate AT/GC ratio. Takes code from #4 and counts A, T, C and G. Scale sets the number of decimal points. Calculates (A + T) / (G + C) rather than the GC fraction.
echo "scale=4; ($(tail -n +2 ../data/fasta/e_coli.fasta |tr -d '\n'| grep -o A | wc -l) + $(tail -n +2 ../data/fasta/e_coli.fasta |tr -d '\n'| grep -o T | wc -l)) / ($(tail -n +2 ../data/fasta/e_coli.fasta |tr -d '\n'| grep -o C | wc -l) + $(tail -n +2 ../data/fasta/e_coli.fasta |tr -d '\n'| grep -o G | wc -l))" | bc
